/*
Представлен алгоритм для определения “зеркальности” матрицы размером n на n.
Вычислите его сложность, используя изученные формулы, в качестве размера входных данных примите n.
*/

func IsReflectMatrix(a [][]int) bool {
	n := len(a)                             // O(1) единичная операция получения длины массива на первом уровне
	if n == 0 {                             // O(1) сравнение происходит один раз
		return true
	}

	for i := 0; i < n-1; i++ {              // O(n-1) раз выполняется внешний цикл
		if len(a[i]) != n {                 // O(1) операция сравнения (!=) будет выполняться в каждом цикле, т.е. (n-1) раз
			return false
		}
		for j := 0; j < n; j++ {            // O(n) раз выполнится внутренний цикл
			if a[i][j] != a[j][i] {         // O(1) операция сравнения (!=) будет выполняться в каждом цикле, т.е. n раз
				return false
			}
		}
	}
	return true
}

1. размер входных данных - n;
2. основная операция - сравнение (!=);
3. в худшем случае число операций сравнения для каждого цикла будет равняться соответственно (n-1) и n;
4. O(n) = O(1) + O(1) + O(n-1)(O(1) + O(n) * O(1)) = O(n-1) * O(n) = O(n^2)
    т.к. в данном случае можно принебречь О(1) как константой.

Ответ: сложность данного алгоритма O(n^2)
