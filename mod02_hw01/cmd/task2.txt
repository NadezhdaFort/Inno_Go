/*
Представлен рекурсивный алгоритм для поиска наименьшего элемента слайса.
Вычислите его сложность, используя изученные формулы.
*/

func MinEl(a []int) int {           // примем n равным len(a) для краткости записи
	// только для первичной проверки
	if len(a) == 0 {                // O(1) - нахождение длины массива и сравнение
		return 0
	}
	if len(a) == 1 {                // O(1) - нахождение длины массива и сравнение
		return a[0]
	}
	t := MinEl(a[:len(a)-1])        // это рекурсивный вызов, который будет вызываться (n-1) раз
	if t <= a[len(a)-1] {           // O(1) - операция сравнения, выполняется после получения ответа при выходе из каждого рекурсивного вызова
		return t
	}
	return a[len(a)-1]
}

1. размер входных данных - len(a) - пусть будет n;
2. основная операция - сравнение (==, <=);
3. число выполняемых основных операций зависит от n, т.к. постоянно происходит сравнение именно с длиной массива;
4. Термины:
     T(n) - время решения задачи, размером n;

    Рекурентное выражение для данного алгоритма:
        T(n) = T(n-1) + 1, т.к.
            каждый вызов рекурсивной функции вызывается на массиве меньшем на 1 от предыдущего
            + 1, т.к. после каждого вызова рекурсивной функции происходит константная операция сравнения.
    Поскольку выход из рекурсивных методов происходит при n=1, то вторым начальным условием будет:
        T(1) = 0
    Применим метод обратной подстановки:
        T(n) = T(n-1) + 1                                       подставляем T(n-1) = T(n-2) + 1,
        T(n) = (T(n-2) + 1) + 1 = T(n-2) + 2                    подставляем T(n-2) = T(n-3) + 1,
        T(n) = (T(n-3) + 1) + 2 = T(n-3) + 3

    Можно заметить закономерность. И чтобы получить обобщенную формулу решения, подставим вместо номера вызова рекурсивной
    функции букву "i", получим:
        T(n) = T(n-i) + i

    Поскольку начальные условия заданы для n=1, то, чтобы получить соответствующую этому значению строку решения рекурсивного
    отношения, нужно подставить i = n-1, в результате получим:
        T(n) = T(n-(n-1)) + n-1 = T(1) + n - 1 = n - 1
    T.к. при оценке сложности алгоритмов пренебрегают константами, то сложность данного алгоритма будет линейной.

Ответ: сложность данного алгоритма O(n)
